# Private preprocessor defines for fixed-point Opus with ARM optimizations
zephyr_compile_options_ifdef(CONFIG_OPUS_CODEC
    -DOPUS_BUILD
    -DUSE_ALLOCA=1
    -DVAR_ARRAYS=0
    -DNONTHREADSAFE_PSEUDOSTACK=0
    -DFIXED_POINT=1
    -DFLOAT=0
    -DALLOW_ALLOCA=1
    -DDISABLE_FLOAT_API=1
    -DHAVE_LRINTF=1
    -DOPUS_ARM_INLINE_EDSP
    -DOPUS_ARM_INLINE_MEDIA
    -DOPUS_ARM_MAY_HAVE_EDSP
    -DOPUS_ARM_PRESUME_EDSP
)

# Compile options to suppress warnings from upstream code

    -Wno-array-bounds
    -Wno-double-promotion
    -Wno-unused-variable
